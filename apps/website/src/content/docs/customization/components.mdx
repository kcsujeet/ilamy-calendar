Out of the box, `@ilamy/calendar` renders accessible native HTML (`<dialog>`, `<button>`, `<input>`, `<select>`) across **React, Vue, Svelte, Solid, and Angular**. When you want your own design-system dialogs, toolbars, or event cards, use **region-level `slots`** (and **`slotProps`** for CSS classes/tokens — following the MUI v6+ convention).

## Supported Region `slots`

| Slot name | Replaces | Works in |
|---|---|---|
| `header` | Full top navigation toolbar & view switcher | React, Vue (`#header`), Svelte, Solid, Angular |
| `event` | Event bar interior content | React, Vue (`#event`), Svelte, Solid, Angular |
| `resource` | Resource header / lane label | React, Vue (`#resource`), Svelte, Solid, Angular |
| `eventForm` | Create / edit / delete event modal | React, Vue (`#event-form`), Svelte, Solid, Angular |
| `moreEventsDialog` | "+N more" overflow dialog | React, Vue (`#more-events-dialog`), Svelte, Solid, Angular |
| `recurrenceScopeDialog` | "Edit recurring event" scope prompt | React, Vue (`#recurrence-scope-dialog`), Svelte, Solid, Angular |

Each slot host stays mounted across navigation, drags, and event updates — inputs keep focus and third-party dialog/select portals keep their state and Escape dismissal order.

## shadcn / Base UI or Radix `Dialog` (React)

Keep your page styling (`@ilamy/calendar/styles.css` or `unstyled`) and swap in your own `Dialog`, `Select`, and `Popover` inside `slots.eventForm`. React Context (your theme store and `useIlamyCalendarContext()`) flows directly into every slot portal:

```tsx title="BaseUiEventFormCalendar.tsx"
import { Dialog } from '@base-ui/react/dialog'
import type { EventFormSlotContext } from '@ilamy/calendar'
import { IlamyCalendar } from '@ilamy/calendar/react'
import '@ilamy/calendar/styles.css'
import { useEffect, useState } from 'react'

function BaseUiEventForm({
  open,
  event,
  initialStart,
  initialEnd,
  initialAllDay,
  onAdd,
  onUpdate,
  onClose,
}: EventFormSlotContext) {
  const [title, setTitle] = useState('')

  useEffect(() => {
    if (open) setTitle(event?.title ?? '')
  }, [open, event])

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    const payload = {
      id: event?.id ?? crypto.randomUUID(),
      title: title.trim() || 'Untitled Event',
      start: event?.start ?? initialStart,
      end: event?.end ?? initialEnd,
      allDay: event?.allDay ?? initialAllDay,
    }
    if (event) onUpdate(event.id, payload)
    else onAdd(payload)
    onClose()
  }

  return (
    <Dialog.Root open={open} onOpenChange={(next) => !next && onClose()}>
      <Dialog.Portal>
        <Dialog.Backdrop className="fixed inset-0 z-40 bg-black/40" />
        <Dialog.Popup className="fixed left-1/2 top-1/2 z-50 w-full max-w-md -translate-x-1/2 -translate-y-1/2 rounded-lg border bg-background p-6 shadow-lg">
          <Dialog.Title className="text-base font-semibold">
            {event ? 'Edit Event' : 'Create Event'}
          </Dialog.Title>
          <form onSubmit={handleSubmit} className="mt-4 space-y-4">
            <input
              aria-label="Title"
              value={title}
              onChange={(e) => setTitle(e.target.value)}
              className="w-full rounded-md border px-3 py-2 text-sm"
            />
            <div className="flex justify-end gap-2">
              <button type="button" onClick={onClose}>Cancel</button>
              <button type="submit">{event ? 'Update' : 'Create'}</button>
            </div>
          </form>
        </Dialog.Popup>
      </Dialog.Portal>
    </Dialog.Root>
  )
}

export function App({ events, onEventAdd, onEventUpdate }) {
  return (
    <IlamyCalendar
      events={events}
      editable
      onEventAdd={onEventAdd}
      onEventUpdate={onEventUpdate}
      slots={{ eventForm: BaseUiEventForm }}
    />
  )
}
```

The same pattern applies to `@radix-ui/react-dialog` (`@/components/ui/dialog`).

## Material UI `Dialog` (React)

Pair MUI components in `slots.eventForm` with any CSS option (for example Plain CSS with `muiThemeToSlotProps(theme)` from [Styling & Themes](/docs/customization/styling#material-ui-theme-recipe)):

```tsx title="MuiEventForm.tsx"
import Button from '@mui/material/Button'
import Dialog from '@mui/material/Dialog'
import DialogActions from '@mui/material/DialogActions'
import DialogContent from '@mui/material/DialogContent'
import DialogTitle from '@mui/material/DialogTitle'
import TextField from '@mui/material/TextField'
import type { EventFormSlotContext } from '@ilamy/calendar'
import { useEffect, useState } from 'react'

export function MuiEventForm({
  open,
  event,
  initialStart,
  initialEnd,
  initialAllDay,
  onAdd,
  onUpdate,
  onClose,
}: EventFormSlotContext) {
  const [title, setTitle] = useState('')

  useEffect(() => {
    if (open) setTitle(event?.title ?? '')
  }, [open, event])

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    const payload = {
      id: event?.id ?? `mui-${Date.now()}`,
      title: title.trim() || 'Untitled Event',
      start: event?.start ?? initialStart,
      end: event?.end ?? initialEnd,
      allDay: event?.allDay ?? initialAllDay,
    }
    if (event) onUpdate(event.id, payload)
    else onAdd(payload)
    onClose()
  }

  return (
    <Dialog open={open} onClose={onClose} fullWidth maxWidth="xs">
      <DialogTitle>{event ? 'Edit Event' : 'Create Event'}</DialogTitle>
      <form onSubmit={handleSubmit}>
        <DialogContent>
          <TextField
            autoFocus
            fullWidth
            label="Title"
            size="small"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
          />
        </DialogContent>
        <DialogActions>
          <Button onClick={onClose}>Cancel</Button>
          <Button type="submit" variant="contained">
            {event ? 'Update' : 'Create'}
          </Button>
        </DialogActions>
      </form>
    </Dialog>
  )
}
```

## bits-ui / shadcn-svelte `Dialog` (Svelte 5)

Pass Svelte 5 components directly into `slots` (plus `getAllContexts()` when the slot reads parent Svelte contexts):

```svelte title="SvelteCalendar.svelte"
<script lang="ts">
  import '@ilamy/calendar/styles.css'
  import { ilamyCalendar } from '@ilamy/calendar/svelte'
  import { getAllContexts } from 'svelte'
  import BitsEventForm from './BitsEventForm.svelte'

  const context = getAllContexts()
  let events = $state(initialEvents)
</script>

<div
  class="h-[600px]"
  {@attach ilamyCalendar({
    events,
    editable: true,
    context,
    slots: { eventForm: BitsEventForm },
    onEventAdd: (added) => (events = [...events, added]),
    onEventUpdate: (updated) =>
      (events = events.map((e) => (e.id === updated.id ? updated : e))),
  })}
/>
```

## Vue 3.5 Scoped Slots

Vue users can supply scoped region templates directly inside `<IlamyCalendar>` (works with native HTML, reka-ui, or shadcn-vue):

```vue title="VueCalendar.vue"
<script setup lang="ts">
import '@ilamy/calendar/styles.css'
import { IlamyCalendar } from '@ilamy/calendar/vue'
import { ref } from 'vue'

const events = ref(initialEvents)
</script>

<template>
  <IlamyCalendar :events="events" editable>
    <template #event="{ event }">
      <span>{{ event.title }}</span>
    </template>
    <template #event-form="{ open, event, initialStart, initialEnd, onAdd, onClose }">
      <MyVueDialog :open="open" @close="onClose" @save="(data) => { ... }" />
    </template>
  </IlamyCalendar>
</template>
```

## React v3 Convenience Props

In `@ilamy/calendar/react`, `renderEvent`, `headerComponent`, `renderResource`, `renderEventForm`, and `onMoreEventsClick` continue to work as thin convenience props mapped onto `slots` alongside `useIlamyCalendarContext()` (for example reading `currentDate`, `prevPeriod`, `nextPeriod`, or `view` inside a custom header or event). For new cross-framework code or custom `moreEventsDialog` / `recurrenceScopeDialog` overrides, prefer `slots`.
